### 1.3 新中间件：TokenBudget

**新文件**：`agent/middlewares/token_budget/core.py`

对标 `IterationBudget`（`agent/middlewares/iteration_budget/core.py`），用 `wrap_model_call` / `awrap_model_call`：

```python
class TokenBudget(AgentMiddleware):
    """Per-session token budget cap. Short-circuits model call when cap is hit."""

    async def awrap_model_call(self, request, handler):
        session_id = _extract_session_id(request)
        cap = state_register_mem.get_state(session_id, StateKey.TOKEN_BUDGET_CAP, 0)
        if cap and cap > 0:
            used = state_register_mem.get_state(session_id, StateKey.TOKEN_BUDGET_USED, 0)
            if used >= cap:
                # 上限触达 → 短路 + WS 通知
                state_register_mem.set_state(session_id, StateKey.TOKEN_BUDGET_LIMITED, True)
                await _push_budget_limited(session_id, used, cap)
                return AIMessage(content=(
                    f"[SYSTEM] Token budget exhausted ({used}/{cap} tokens). "
                    f"Stop calling tools and provide a summary. "
                    f"The user can reset the budget by sending a new message."
                ))
        # 正常执行
        result = await handler(request)
        # 后置：从 result 提取 usage_metadata，累加
        await self._accumulate_usage(session_id, result)
        return result

    async def _accumulate_usage(self, session_id, result):
        """从 AIMessage.usage_metadata 提取 token 数，累加到 state_register。

        Cost 查找优先级（§1.8 四层方案）：
          C. provider 直返 cost（usage.cost 字段，最准，零维护）→
          MODEL_PRICING_TABLE（用户 .models 完整表，sherry .models.example 给默认值）→
          B. litellm model_cost JSON 缓存（社区维护，定期 sync）→
          _default（MODEL_PRICING_TABLE 里的 _default 字段）
        """
        ai_msg = _extract_ai_message(result)  # 复用 llm_retry/core.py 的提取模式
        if ai_msg is None:
            return
        usage = getattr(ai_msg, "usage_metadata", None)
        if not usage:
            return
        input_tokens = int(usage.get("input_tokens", 0))
        output_tokens = int(usage.get("output_tokens", 0))
        reasoning = int((usage.get("output_token_details") or {}).get("reasoning_tokens", 0))
        total = input_tokens + output_tokens

        # 累加 token 数（token 数是准的，不受 cost 查找影响）
        with _budget_lock(session_id):
            used = state_register_mem.get_state(session_id, StateKey.TOKEN_BUDGET_USED, 0)
            state_register_mem.set_state(session_id, StateKey.TOKEN_BUDGET_USED, used + total)

            # ★ C: provider 直返 cost（OpenRouter / 某些 OpenAI 兼容 API 在 usage 里带 cost）
            provider_cost = _extract_provider_cost(usage)
            if provider_cost is not None:
                cost_delta = provider_cost  # provider 已经算好，直接用
            else:
                # 用户表/B/_default 三级 fallback（见 §1.8 _resolve_model_pricing）
                pricing = _resolve_model_pricing(ai_msg)
                cost_delta = (input_tokens * pricing["input"] + output_tokens * pricing["output"]) / 1_000_000

            cost = state_register_mem.get_state(session_id, StateKey.TOKEN_BUDGET_COST, 0.0)
            state_register_mem.set_state(session_id, StateKey.TOKEN_BUDGET_COST, round(cost + cost_delta, 6))

        # 预警阈值（MODEL_PRICING["budget_warn_threshold"] = 0.80）
        cap = state_register_mem.get_state(session_id, StateKey.TOKEN_BUDGET_CAP, 0)
        if cap and cap > 0:
            new_used = used + total
            if new_used >= cap * MODEL_PRICING["budget_warn_threshold"] and not _warned:
                await _push_budget_warning(session_id, new_used, cap)


def _extract_provider_cost(usage: dict) -> float | None:
    """C: provider 直返 cost 检查。

    某些 provider（OpenRouter / 某些 OpenAI 兼容 API）在 usage 里直接返回 cost，
    provider 自己算的最准。检查可能的字段名：
      - usage.cost（OpenRouter 风格）
      - usage.total_cost / usage.prompt_cost / usage.completion_cost
      - usage estimated_cost
    """
    for field in ("cost", "total_cost", "estimated_cost"):
        value = usage.get(field)
        if isinstance(value, (int, float)) and value > 0:
            return float(value)
    # OpenRouter 把 cost 拆成 prompt + completion
    prompt_cost = usage.get("prompt_cost")
    completion_cost = usage.get("completion_cost")
    if isinstance(prompt_cost, (int, float)) and isinstance(completion_cost, (int, float)):
        return float(prompt_cost) + float(completion_cost)
    return None


def _resolve_model_pricing(ai_msg) -> dict[str, float]:
    """用户表/B/_default 三级 fallback 查找（§1.8）。

    优先级（从高到低）：
      1. MODEL_PRICING["model_pricing_per_m_tokens"] dict（从用户 .models 的
         MODEL_PRICING_TABLE 加载，sherry .models.example 给默认值——见 §1.8.2）
      2. B: litellm model_cost JSON 缓存（启动时 _load_litellm_prices() 加载到 _LITELLM_CACHE）
      3. _default（MODEL_PRICING dict 里的 "_default" 字段，默认 input $0.5/M / output $1.5/M）

    返回 {"input": float, "output": float}（USD per 1M tokens）。
    """
    model_name = _extract_model_name(ai_msg)

    # 1. 用户完整表（hydrate 时已从 .models 加载到 MODEL_PRICING dict）
    # 见 §1.8.2：MODEL_PRICING_TABLE 是用户 .models 的完整价格表，sherry .models.example
    # 给默认值（国产模型推荐配置），hydrate 时注入 os.environ，model_pricing.py 模块
    # 加载时读 env 构造 MODEL_PRICING dict。所以查这一个 dict 就够。
    pricing_table = MODEL_PRICING["model_pricing_per_m_tokens"]
    if model_name in pricing_table:
        return pricing_table[model_name]

    # 2. B: litellm JSON 缓存（启动时 _load_litellm_prices() 加载到 _LITELLM_CACHE）
    litellm_entry = _LITELLM_CACHE.get(model_name)
    if litellm_entry is not None:
        return litellm_entry

    # 3. _default 兜底（用户表里的 _default 字段，sherry .models.example 默认 input $0.5/M / output $1.5/M）
    return pricing_table["_default"]
```

### 1.8 计费表四层维护方案（C + 用户完整表 + B + _default）

**问题**：sherry 当前 `MODEL_PRICING` 硬编码只列了 3 个模型（glm-5 / deepseek-chat / kimi-latest），用户用很多模型时大部分掉到 `_default`（input $1/M / output $3/M），估算误差大，budget warning 不准。而且用户不能改价格表（要改代码）。

**方案**：把价格表整个搬到 `.models`，用户可配置。四层组合，让计费表维护成本接近零，同时精度可控。优先级（从高到低）：

| 层 | 来源 | 维护成本 | 精度 | 适用场景 |
|---|---|---|---|---|
| **C** | provider 直返 `usage.cost` | 零（provider 算） | 最高（provider 自己最准） | OpenRouter / 某些 OpenAI 兼容 API |
| **用户完整表** | `.models` 里 `MODEL_PRICING_TABLE`（sherry `.models.example` 给默认值） | sherry 团队改 .models.example + 用户可覆盖 | 高（sherry .models.example 推荐配置 + 用户自定义） | 默认开箱即用 + 用户精确控制 |
| **B** | litellm `model_cost` JSON 缓存 | 低（社区维护，定期 sync） | 高（100+ 模型覆盖） | 用户表没覆盖的模型 |
| **`_default`** | `MODEL_PRICING_TABLE` 里的 `_default` 字段 | 零（用户表里配） | 低（±50% 误差，估算用） | 完全没覆盖的模型 |

**Cost 查找流程**（`_accumulate_usage` §1.3）：

```
1. C: 检查 usage.cost / usage.total_cost / usage.prompt_cost + completion_cost
   ├─ 有 → 直接用 provider 算好的 cost（最准，零维护）
   └─ 没有 → 继续
2. 查 _resolve_model_pricing(ai_msg) 三级 fallback:
   a. MODEL_PRICING["model_pricing_per_m_tokens"] dict（从用户 .models 的
      MODEL_PRICING_TABLE 加载，sherry .models.example 给默认值）
   b. B: litellm JSON 缓存（启动时加载到 _LITELLM_CACHE）
   c. _default（用户表里的 _default 字段）
   返回 {input, output} 价格 → 自己算 cost_delta
```

**文件**：`agent/middlewares/token_budget/core.py`（§1.3 `_extract_provider_cost`）

**`config/features/infra_side/model_pricing.py`** 改造：

```python
"""Per-model token pricing — 完全用户可配置（从 .models 加载）。

sherry 代码层不硬编码任何具体模型价格，只保留 _FALLBACK_DEFAULT 作 hydrate 失败兜底。
用户在 .models 的 MODEL_PRICING_TABLE 字段配置完整价格表，
sherry .models.example 提供合理默认值（国产模型推荐配置）。
"""

import json
import os
from typing import TypedDict

# 仅作 hydrate 失败时的最终兜底（代码层硬编码，不暴露给用户）
_FALLBACK_DEFAULT = {"input": 0.5, "output": 1.5}


class ModelPricingConfig(TypedDict):
    model_pricing_per_m_tokens: dict[str, dict[str, float]]
    budget_warn_threshold: float


def _load_pricing_table() -> dict[str, dict[str, float]]:
    """从 MODEL_PRICING_TABLE env 加载用户配置的价格表。

    hydrate 时 config/models_settings.py 把 .models 的 MODEL_PRICING_TABLE
    注入 os.environ（跟其他 .models 字段同模式）。
    缺失或解析失败 → 只返回 _FALLBACK_DEFAULT。
    """
    raw = os.getenv("MODEL_PRICING_TABLE", "").strip()
    if not raw:
        return {"_default": _FALLBACK_DEFAULT}
    try:
        table = json.loads(raw)
        if not isinstance(table, dict):
            return {"_default": _FALLBACK_DEFAULT}
        # 确保 _default 存在
        if "_default" not in table:
            table["_default"] = _FALLBACK_DEFAULT
        return table
    except Exception:
        return {"_default": _FALLBACK_DEFAULT}


MODEL_PRICING: ModelPricingConfig = {
    "model_pricing_per_m_tokens": _load_pricing_table(),
    "budget_warn_threshold": 0.80,
}
```

**`config/models_settings.py::hydrate()`** 加载逻辑：

```python

    sherry .models.example 提供默认值（国产模型推荐配置），用户可在 .models 改。
    hydrate 后 config/features/infra_side/model_pricing.py 模块加载时读 env 构造 dict。
    """

```

**关键设计**：

1. **sherry 代码层零具体模型价格硬编码**——价格表完全从 `.models` 读
2. **`_FALLBACK_DEFAULT` 仅 hydrate 失败时兜底**——正常情况下用户表里的 `_default` 是主兜底
3. **用户可改/加/删模型**——不需要改代码，改 `.models` 就行
4. **升级时 `.models.example` 更新默认值，用户已有 `.models` 不被覆盖**（跟其他 .models 字段同语义）
5. **`_default` 也是表的一部分，用户可改**——`MODEL_PRICING_TABLE._default` 字段用户可调

**验收**：
- 用户配 `MODEL_PRICING_TABLE={"claude-3-5-sonnet": {"input": 3.0, "output": 15.0}}` → 该模型 cost 用用户价格算
- 用户改 `_default` → 完全没覆盖的模型用新 _default 估算
- hydrate 失败 → `_load_pricing_table` 返回 `_FALLBACK_DEFAULT`

```python
import json
from pathlib import Path
from config.path import SRC_DIR

_LITELLM_CACHE: dict[str, dict[str, float]] = {}

def _load_litellm_prices() -> None:
    """B: 启动时从本地缓存加载 litellm model_cost JSON 到 _LITELLM_CACHE。

    缓存文件路径：SRC_DIR / "data" / "litellm_prices.json"
    缺失时 _LITELLM_CACHE 为空，_resolve_model_pricing 走 _default 兜底。
    """
    cache_path = SRC_DIR / "data" / "litellm_prices.json"
    if not cache_path.exists():
        return
    try:
        data = json.loads(cache_path.read_text(encoding="utf-8"))
        for model_name, entry in data.items():
            if model_name == "sample_spec" or not isinstance(entry, dict):
                continue
            input_price = entry.get("input_cost_per_token")
            output_price = entry.get("output_cost_per_token")
            if input_price is not None and output_price is not None:
                # litellm 用 per-token，sherry 用 per-1M-tokens，转换
                _LITELLM_CACHE[model_name] = {
                    "input": float(input_price) * 1_000_000,
                    "output": float(output_price) * 1_000_000,
                }
    except Exception as exc:
        logger.warning("litellm prices load failed: {}", exc)

_load_litellm_prices()  # 模块加载时执行
```

#### 1.8.4 优先级总结（四层）

```
_accumulate_usage(session_id, result):
  usage = extract usage_metadata from ai_msg
  input_tokens, output_tokens = usage.input/output_tokens

  # token 数累加（准的，不受 cost 查找影响）
  TOKEN_BUDGET_USED += input_tokens + output_tokens

  # cost 累加（估算，按优先级查）
  ★ C: provider 直返 cost（usage.cost / total_cost / prompt+completion_cost）
     ├─ 命中 → cost_delta = provider_cost（最准）
     └─ 不命中 →
       _resolve_model_pricing(ai_msg) 三级 fallback:
         用户表: MODEL_PRICING dict（hydrate 时从 .models 的 MODEL_PRICING_TABLE 加载，
                sherry .models.example 给默认值）→ 命中 → 用
         B: _LITELLM_CACHE（litellm JSON 缓存）→ 命中 → 用
         _default: 用户表里的 "_default" 字段 → 兜底
       cost_delta = (input × pricing.input + output × pricing.output) / 1_000_000

  TOKEN_BUDGET_COST += cost_delta
```

**关键洞察**：token 数始终是准的（provider 直接报），cost 是估算的（按优先级查表）。budget 封顶功能本身不受 cost 估算误差影响——`TOKEN_BUDGET_USED` 用 token 数判阈值，`TOKEN_BUDGET_COST` 只是给用户看的估算。

**优先级链简化**（从五层简化到四层）：

| 改造前（五层） | 改造后（四层） |
|---|---|
| C: provider 直返 | C: provider 直返（不变） |
| D: 用户 override（增量覆盖） | **用户完整表 `MODEL_PRICING_TABLE`**（替换 D + sherry 内置两层） |
| sherry 内置 dict（硬编码） | ↑ 合并 |
| B: litellm 缓存 | B: litellm 缓存（不变） |
| _default | _default（从用户表读，不硬编码） |
