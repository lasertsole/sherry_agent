# I18n 国际化修改说明书

> 本文档包含 i18n 国际化改造的完整修改细节，任何 Agent 可据此在旧版本上精确复现全部改动。
> 每处修改均标注：文件路径、旧代码（完整上下文）、新代码（完整上下文）、修改原因。
> 路径均相对于项目根目录 `sherry_agent`。
>
> **前提条件**：项目已通过 `nuxt.config.ts` 配置了 `@nuxtjs/i18n` 模块，locale 文件目录为 `app/i18n/locales/`，默认语言 `zh`，策略 `prefix_except_default`。在 Nuxt 运行时中 `useI18n()` / `$t()` 为自动导入，无需额外注册。但测试环境（vitest）中需要显式 `import { useI18n } from 'vue-i18n'`。

**原因**：locale 文件一致性测试，不需要 vue-i18n 依赖，纯 JSON 校验。测试 3 类问题：key 对齐、占位符一致、无空值。

```typescript
import { describe, it, expect } from 'vitest';
import zh from '../locales/zh.json';
import en from '../locales/en.json';

/** Recursively collect all leaf key paths from a nested object, e.g. "chatInput.send" */
function collectKeys(obj: Record<string, unknown>, prefix = ''): string[] {
  const keys: string[] = [];
  for (const [key, value] of Object.entries(obj)) {
    const path = prefix ? `${prefix}.${key}` : key;
    if (value !== null && typeof value === 'object' && !Array.isArray(value)) {
      keys.push(...collectKeys(value as Record<string, unknown>, path));
    } else {
      keys.push(path);
    }
  }
  return keys;
}

/** Recursively get the value at a dot-separated path, e.g. get("chatInput.send", obj) */
function getValue(path: string, obj: Record<string, unknown>): unknown {
  return path.split('.').reduce<unknown>((acc, key) => {
    if (acc !== null && typeof acc === 'object') {
      return (acc as Record<string, unknown>)[key];
    }
    return undefined;
  }, obj);
}

const zhKeys = collectKeys(zh);
const enKeys = collectKeys(en);

describe('i18n locale files', () => {
  describe('key consistency', () => {
    it('zh.json and en.json have the same number of keys', () => {
      expect(zhKeys.length).toBe(enKeys.length);
    });

    it('every key in zh.json exists in en.json', () => {
      const enSet = new Set(enKeys);
      const missing = zhKeys.filter((k) => !enSet.has(k));
      expect(missing).toEqual([]);
    });

    it('every key in en.json exists in zh.json', () => {
      const zhSet = new Set(zhKeys);
      const missing = enKeys.filter((k) => !zhSet.has(k));
      expect(missing).toEqual([]);
    });
  });

  describe('value sanity', () => {
    it('no empty string values in zh.json', () => {
      const empty = zhKeys.filter((k) => getValue(k, zh) === '');
      expect(empty).toEqual([]);
    });

    it('no empty string values in en.json', () => {
      const empty = enKeys.filter((k) => getValue(k, en) === '');
      expect(empty).toEqual([]);
    });

    it('placeholder tokens match between locales', () => {
      const tokenRegex = /\{(\w+)\}/g;
      for (const key of zhKeys) {
        const zhVal = String(getValue(key, zh));
        const enVal = String(getValue(key, en));
        const zhTokens = (zhVal.match(tokenRegex) ?? []).sort();
        const enTokens = (enVal.match(tokenRegex) ?? []).sort();
        expect(enTokens).toEqual(zhTokens);
      }
    });
  });

  describe('known keys', () => {
    it('contains chatInput.send', () => {
      expect(zhKeys).toContain('chatInput.send');
      expect(enKeys).toContain('chatInput.send');
    });

    it('contains history.noSessions', () => {
      expect(zhKeys).toContain('history.noSessions');
      expect(enKeys).toContain('history.noSessions');
    });

    it('contains errors.replyFailed with {reason} placeholder', () => {
      expect(zhKeys).toContain('errors.replyFailed');
      const zhVal = getValue('errors.replyFailed', zh) as string;
      const enVal = getValue('errors.replyFailed', en) as string;
      expect(zhVal).toContain('{reason}');
      expect(enVal).toContain('{reason}');
    });

    it('contains imagePreview.hint', () => {
      expect(zhKeys).toContain('imagePreview.hint');
      expect(enKeys).toContain('imagePreview.hint');
    });
  });
});
```
