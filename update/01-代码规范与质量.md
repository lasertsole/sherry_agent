## 1. 整体架构

```
项目根目录/
├── eslint.config.mjs          # ESLint 9 flat config（或 .eslintrc.js 旧版）
├── prettier.config.mjs        # Prettier 配置（或 .prettierrc）
├── .prettierignore            # Prettier 忽略文件
├── tsconfig.json              # TypeScript 配置
├── .vscode/settings.json      # VS Code 工作区配置（format-on-save + eslint auto-fix）
├── package.json               # 依赖声明 + lint-staged 配置
├── .husky/pre-commit          # Git pre-commit 钩子
└── patches/                   # patch-package 补丁目录
```

**工具链全景**：

```
ESLint（代码质量+风格检查）
  └── eslint-plugin-prettier（将 Prettier 规则作为 ESLint 规则运行，避免冲突）
  └── typescript-eslint（Type语法检查）
  └── eslint-plugin-vue（Vue SFC 检查）[Vue 项目适用]
Prettier（代码格式化）
lint-staged（仅检查暂存文件）
husky（Git 钩子触发 lint-staged）
patch-package（依赖补丁）
```

## 2. ESLint 9 Flat Config 完整配置

### 2.1 安装依赖

```bash
# ESLint 9 + 核心插件
npm install -D eslint @eslint/js globals

# TypeScript 支持
npm install -D typescript-eslint @typescript-eslint/parser @typescript-eslint/eslint-plugin
```

### 2.2 配置文件 `eslint.config.mjs`

> **适用场景**：Vue 3 + TypeScript 项目。React/纯 TS 项目删除 Vue 相关部分即可。

```javascript
// eslint.config.mjs
import { defineConfig, globalIgnores } from "eslint/config";
import js from "@eslint/js";
import globals from "globals";
import pluginVue from "eslint-plugin-vue";
import tseslint from "typescript-eslint";
import eslintPluginPrettierRecommended from "eslint-plugin-prettier/recommended";
import vueParser from "vue-eslint-parser";
import tsParser from "@typescript-eslint/parser";
import importPlugin from "eslint-plugin-import";

export default defineConfig([
  // ============ 1. 全局忽略 ============
  globalIgnores(
    [
      "dist",
      "static",
      ".vscode",
      ".husky",
      ".card-cli",
      ".cache",
      ".cloudbuild",
      "node_modules",
      "coverage",
    ],
    "global ignores config",
  ),

  // ============ 2. 基础 JS 推荐配置 ============
  js.configs.recommended,

  // ============ 3. 全局插件注册 ============
  {
    plugins: { vue: pluginVue, import: importPlugin },
  },

  // ============ 4. 语言选项与全局变量 ============
  {
    languageOptions: {
      sourceType: "module",
      globals: {
        ...globals.browser,
        ...globals.node,

        // —— 按需声明项目全局变量 ——
        // 以下是示例：平台 SDK / 框架全局对象
        // 纯 Web 项目可删除这些，只保留 browser + node
        ENV: "readonly",
        ENV_MODE: "readonly",

        // Vue 3 编译器宏（Vue 项目必须）
        withDefaults: "readonly",
        defineProps: "readonly",
        defineEmits: "readonly",
        defineModel: "readonly",
        defineExpose: "readonly",
        defineOptions: "readonly",
        defineSlots: "readonly",
      },
    },
  },

  // ============ 5. TypeScript 推荐配置 ============
  tseslint.configs.recommended,

  // ============ 6. Vue 3 推荐配置（Vue 项目需要） ============
  ...pluginVue.configs["flat/recommended"],
  {
    files: ["**/*.vue"],
    languageOptions: {
      parser: vueParser,
      parserOptions: {
        parser: tsParser, // <script lang="ts"> 块使用 TS 解析器
        extraFileExtensions: [".vue"],

    },

  // ============ 7. Prettier 集成 ============

  eslintPluginPrettierRecommended,

  // ============ 8. 自定义规则 ============
  {
    rules: {
      // —— 严格规则（error 级别）——
      "import/first": 2, // import 必须在文件顶部
      semi: 2, // 强制分号
      "vue/no-mutating-props": 2, // 禁止修改 props
      "vue/no-v-html": 2, // 禁止 v-html（防 XSS）
      "vue/no-unused-vars": 2, // 禁止未使用的变量
      "@typescript-eslint/no-unused-vars": 2, // TS 未使用变量
      "@typescript-eslint/no-explicit-any": 2, // 禁止 any 类型
      "@typescript-eslint/explicit-member-accessibility": 2, // 类成员必须显式声明可见性
      curly: ["error", "all"], // if/for/while 体必须加 {}

      // —— 警告规则（warn 级别）——
      "no-console": 1, // console 警告
      eqeqeq: [1, "smart"], // 鼓励 ===（null/undefined 例外）
      "@typescript-eslint/naming-convention": [
        1,
        { selector: "class", format: ["PascalCase"] },
      ],
      "@typescript-eslint/no-this-alias": 1,
      "@typescript-eslint/no-empty-function": 1,
      "@typescript-eslint/no-empty-interface": 1,
      "@typescript-eslint/no-require-imports": 1,
      "vue/multi-word-component-names": 1, // 组件名多单词（warn）
```

```javascript
// prettier.config.mjs
/**
 * @see https://prettier.io/docs/en/configuration.html

 */
export default {
  // 一行最多 120 字符
  printWidth: 120,
  // 使用 2 个空格缩进
  tabWidth: 2,
  // 使用单引号
  singleQuote: true,
  // 语句末尾加分号
  semi: true,
  // 多行逗号分割的语法中，最后一行不加逗号
  trailingComma: "none",
  // JSX 使用单引号

  // 单个参数的箭头函数不加括号 x => x
```

## 6. 环境感知日志工具

### 6.1 设计思路

生产环境禁止输出日志，开发/SIT/UAT 环境允许输出不同级别的日志。非浏览器环境（如原生 SDK 日志）需特殊处理。

### 6.2 完整实现

```typescript
// src/utils/log.ts

// ============ 环境工具（框架无关） ============
// 根据项目实际的环境变量机制调整 ENV_MODE 的来源
// 方式一：webpack DefinePlugin 注入的全局变量
// 方式二：import.meta.env.VITE_ENV_MODE（Vite）
// 方式三：process.env.NODE_ENV

declare const ENV_MODE: string;

export class EnvironmentUtils {
  public static isLocal(): boolean {
    return ENV_MODE === "local";
  }
  public static isSit(): boolean {
    return ENV_MODE === "sit";
  }
  public static isUat(): boolean {
    return ENV_MODE === "uat";
  }
  public static isIcsl(): boolean {
    return ENV_MODE === "icsl";
  }
  public static isProd(): boolean {
    return ENV_MODE === "prod";
  }
  public static isDev(): boolean {
    return this.isLocal() || this.isSit() || this.isUat();
  }
}

// ============ 日志工具 ============
// 六个级别：l(调试数据) / d(debug) / i(info) / w(warn) / e(error) / f(fatal)

// 原生日志接口（可选：如果项目有原生 SDK 日志能力）
// 如果没有原生 SDK，HWH5Log 可以是空函数
interface NativeLogger {
  log(params: { type: string; content: string }): void;
}

function getNativeLogger(): NativeLogger | null {
  // 按需替换为项目的原生日志接口
  // 例如：return (window as any).HWH5 || null;
  return null;
}

function nativeLog(
  type: string,
  tag: string,
  message: string,
  ...args: any[]
): void {
  if (EnvironmentUtils.isLocal()) return; // 本地不走原生日志
  try {
    const logger = getNativeLogger();
    logger?.log({
      type,
      content: `${tag} ${message} ${args.length ? JSON.stringify(args, null, 2) : ""}`,
    });
  } catch (error) {
    if (EnvironmentUtils.isDev()) {
      console.error("Native log error", error);
    }
  }
}

export class Logger {
  /** 调试数据（仅 dev 环境输出到 console） */
  public l(...args: any[]): void {
    if (EnvironmentUtils.isDev()) {
      console.warn("l 调试数据", ...args);
    }
  }

  /** debug 级别（原生日志 + dev 环境 console） */
  public d(tag: string, message: string, ...args: any[]): void {
    nativeLog("d", tag, message, ...args);
    if (EnvironmentUtils.isDev()) {
      console.warn("d", tag, message, ...args);
    }
  }

  /** info 级别 */
  public i(tag: string, message: string, ...args: any[]): void {
    nativeLog("i", tag, message, ...args);
    if (EnvironmentUtils.isDev()) {
      console.warn("i", tag, message, ...args);
    }
  }

  /** warn 级别 */
  public w(tag: string, message: string, ...args: any[]): void {
    nativeLog("w", tag, message, ...args);
    if (EnvironmentUtils.isDev()) {
      console.warn("w", tag, message, ...args);
    }
  }

  /** error 级别（始终输出） */
  public e(tag: string, message: string, ...args: any[]): void {
    nativeLog("e", tag, message, ...args);
    if (EnvironmentUtils.isDev()) {
      console.error(tag, message, ...args);
    }
  }

  /** fatal 级别（仅原生日志，不输出到 console） */
  public f(tag: string, message: string, ...args: any[]): void {
    nativeLog("f", tag, message, ...args);
  }
}

// 单例导出
export const logUtils = new Logger();

// ============ 简化版（无原生 SDK 的纯 Web 项目） ============
// 如果项目不需要原生日志能力，可以用以下简化版

export class SimpleLogger {
  public l(...args: any[]): void {
    if (EnvironmentUtils.isDev()) console.log(...args);
  }
  public w(...args: any[]): void {
    if (EnvironmentUtils.isDev()) console.warn(...args);
  }
  public e(...args: any[]): void {
    if (!EnvironmentUtils.isLocal()) {
      console.error(...args);
    } else {
      console.warn("local环境报错", ...args);
    }
  }
}

export const logUtil = new SimpleLogger();
```

### 6.3 使用方式

```typescript
import { logUtils } from "@/utils/log";

// 调试数据（仅 dev 环境）
logUtils.l("用户数据:", userData);

// 带标签的日志
logUtils.e("UserService", "获取用户信息失败", error);

// 简化版
```

## 7. patch-package 依赖补丁

### 7.1 场景

第三方 npm 包有 bug 但无法等待修复，或者需要对第三方包做定制化修改。

### 7.2 安装与配置

```bash
npm install -D patch-package
```

在 `package.json` 中添加 postinstall 脚本：

```json
{
  "scripts": {
    "postinstall": "patch-package"
  }
}
```

### 7.3 使用流程

```bash
# 1. 修改 node_modules 中的文件
#    例如：node_modules/some-package/dist/index.js

# 2. 生成补丁文件
npx patch-package some-package

# 3. 补丁文件自动生成在 patches/ 目录
#    patches/some-package+1.2.3.patch

# 4. 提交补丁文件到 git
git add patches/
git commit -m "fix: patch some-package for bug fix"
```

### 7.4 补丁文件示例

```
patches/
├── @materials+approve-button-group-vue3+0.0.30.patch
└── @materials+form-render-vue3+0.0.10.patch
```

补丁文件是标准 diff 格式，团队成员 `npm install` 后 `postinstall` 会自动应用所有补丁。

## 8. 异构项目适配指南

### 8.1 React 项目

| 配置项   | 调整内容                                                                                               |
| -------- | ------------------------------------------------------------------------------------------------------ |
| ESLint   | 删除 `eslint-plugin-vue`、`vue-eslint-parser`，添加 `eslint-plugin-react`、`eslint-plugin-react-hooks` |
| Prettier | 配置完全通用，无需修改                                                                                 |
| VS Code  | `[vue]` 改为 `[jsx]`、`[tsx]`，扩展添加 `React.code-react`                                             |
| 路径别名 | `@/*` 仍可使用，`tsconfig.json` 中配置 paths                                                           |
| 日志工具 | 完全通用，调整 `ENV_MODE` 来源即可                                                                     |

### 8.2 Vite 项目

| 配置项       | 调整内容                                                                     |
| ------------ | ---------------------------------------------------------------------------- |
| 环境变量     | `ENV_MODE` 改为通过 `import.meta.env.MODE` 或 `.env` 文件的 `VITE_` 前缀变量 |
| DefinePlugin | Vite 使用 `define` 配置项替代 webpack `DefinePlugin`                         |
| 日志工具     | `EnvironmentUtils` 改为读取 `import.meta.env.MODE`                           |

### 8.3 Angular 项目

| 配置项     | 调整内容                                                      |
| ---------- | ------------------------------------------------------------- |
| ESLint     | 使用 `@angular-eslint/eslint-plugin`，替换 Vue 相关插件       |
| Prettier   | 配置通用，但 `singleAttributePerLine` 可选                    |
| TypeScript | Angular 已内置 TS 支持，`tsconfig.json` 使用 Angular CLI 生成 |

### 8.4 Node.js 后端项目

| 配置项   | 调整内容                                                                    |
| -------- | --------------------------------------------------------------------------- |
| ESLint   | 删除 Vue/浏览器全局变量，保留 `globals.node`                                |
| Prettier | 配置完全通用                                                                |
| 日志工具 | `ENV_MODE` 改为 `process.env.NODE_ENV`，`nativeLog` 替换为 `winston`/`pino` |
